CREATE TABLE IF NOT EXISTS addresses (
    id int8 PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
    street_address varchar(100) NOT NULL CHECK ( char_length(street_address) >= 5 AND char_length(street_address) <= 100),
    apartment varchar(50),
    city varchar(50) NOT NULL CHECK ( char_length(city) >= 2 AND char_length(city) <= 50),
    state varchar(50),
    postal_code varchar(10) NOT NULL CHECK ( char_length(postal_code) >= 3 AND char_length(postal_code) <= 10),
    country_code varchar(2) NOT NULL CHECK ( char_length(country_code) = 2),
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS users (
    id int8 PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
    first_name varchar(50) NOT NULL,
    last_name varchar(50) NOT NULL,
    email text NOT NULL UNIQUE ,
    phone_number varchar(15) NOT NULL CHECK
        ( char_length(phone_number) >= 11 AND char_length(phone_number) <= 15),
    address_id int8 NOT NULL REFERENCES addresses(id),
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_users_address_id ON users(address_id);
